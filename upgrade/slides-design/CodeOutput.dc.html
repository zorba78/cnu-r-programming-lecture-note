프로그램 1 · 사전 할당 
prealloc_vec <- function(n) {
  x <- numeric(n)
  for (i in seq_len(n)) {
    x[i] <- i
  }
  x
}
프로그램 2 · 이어 붙이기 
grow_vec <- function(n) {
  x <- NULL
  for (i in seq_len(n)) {
    x <- c(x, i)
  }
  x
}
출력 · n = 50,000
